# Descriptive Statistics on Iris Dataset

# Step 1: Import Libraries
import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt

# Step 2: Load Dataset
df = sns.load_dataset('iris')

print("First 5 Rows of Dataset:\n")
print(df.head())

# Step 3: Basic Information
print("\nDataset Info:\n")
print(df.info())

# Step 4: Descriptive Statistics

print("\nMean:\n")
print(df.mean(numeric_only=True))

print("\nMedian:\n")
print(df.median(numeric_only=True))

print("\nMinimum Values:\n")
print(df.min(numeric_only=True))

print("\nMaximum Values:\n")
print(df.max(numeric_only=True))

print("\nVariance:\n")
print(df.var(numeric_only=True))

print("\nStandard Deviation:\n")
print(df.std(numeric_only=True))

# Step 5: Summary Statistics
print("\nFull Summary (describe):\n")
print(df.describe())

# Step 6: Group By Species
grouped = df.groupby('species')

print("\nSpecies-wise Mean:\n")
print(grouped.mean())

print("\nDetailed Species-wise Statistics:\n")
print(grouped.agg(['mean', 'median', 'min', 'max', 'std', 'var']))

# Step 7: Visualization (Optional)

# Boxplot for Petal Length by Species
plt.figure()
sns.boxplot(x='species', y='petal_length', data=df)
plt.title("Petal Length Distribution by Species")
plt.show()

# Histogram for Sepal Length
plt.figure()
sns.histplot(df['sepal_length'], kde=True)
plt.title("Sepal Length Distribution")
plt.show()

# Step 8: Conclusion
print("\nConclusion:")
print("Setosa has smallest petals and low variation.")
print("Versicolor has moderate values.")
print("Virginica has largest petals and higher variation.")




Q1: What is descriptive statistics?

👉 It is the method of summarizing and describing data using measures like mean, median, variance, etc.

Q2: What is the mean?

👉 Mean is the average of all values.
Formula: Sum of values / Number of values.

Q3: What is the median?

👉 Median is the middle value when data is arranged in order.

Q4: Difference between mean and median?

👉 Mean is affected by outliers, while median is not.

Q5: What is variance?

👉 Variance measures how far data points are spread from the mean.

Q6: What is standard deviation?

👉 It is the square root of variance and shows data dispersion.

Q7: What are min and max?

👉 Minimum is the smallest value, maximum is the largest value.
