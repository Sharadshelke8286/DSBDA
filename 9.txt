# Data Visualization II - Titanic Dataset (Single Cell)

# -------------------------------
# Step 1: Import Libraries
# -------------------------------
import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt

# -------------------------------
# Step 2: Load Dataset
# -------------------------------
df = sns.load_dataset('titanic')

print("First 5 Rows:\n")
print(df.head())

print("\nDataset Info:\n")
print(df.info())

# -------------------------------
# Step 3: Handle Missing Values
# -------------------------------
df['age'].fillna(df['age'].median(), inplace=True)

print("\nMissing Values After Handling:\n")
print(df.isnull().sum())

# -------------------------------
# Step 4: Boxplot for Age
# -------------------------------
plt.figure(figsize=(8,5))
sns.boxplot(x=df['age'])
plt.title("Boxplot of Age")
plt.xlabel("Age")
plt.show()

# -------------------------------
# Step 5: Age vs Gender vs Survival
# -------------------------------
plt.figure(figsize=(10,6))
sns.boxplot(x='sex', y='age', hue='survived', data=df)

plt.title("Age vs Gender vs Survival")
plt.xlabel("Gender")
plt.ylabel("Age")
plt.show()

# -------------------------------
# Step 6: Identify Outliers (IQR)
# -------------------------------
Q1 = df['age'].quantile(0.25)
Q3 = df['age'].quantile(0.75)
IQR = Q3 - Q1

lower_bound = Q1 - 1.5 * IQR
upper_bound = Q3 + 1.5 * IQR

outliers = df[(df['age'] < lower_bound) | (df['age'] > upper_bound)]

print("\nLower Bound:", lower_bound)
print("Upper Bound:", upper_bound)

print("\nOutliers in Age:\n")
print(outliers[['age']])

# -------------------------------
# Step 7: Observations
# -------------------------------
print("\n--- Observations ---")
print("1. Most passengers are between 20–40 years.")
print("2. Median age is around 28–30.")
print("3. Some high-age values are outliers.")
print("4. Females have higher survival rates.")
print("5. Children show better survival chances.")

# -------------------------------
# Step 8: Inferences
# -------------------------------
print("\n--- Inferences ---")
print("1. Women and children were prioritized.")
print("2. Gender and age strongly affected survival.")
print("3. Adult males had lower survival probability.")
print("4. Age distribution is slightly right-skewed.")

Q4. What is a boxplot?
👉 A boxplot is a graphical representation that shows the distribution of data using median, quartiles, and outliers.

Q5. What does the box in a boxplot represent?
👉 The box represents the Interquartile Range (IQR), i.e., values between Q1 and Q3.

Q6. What is the median in a boxplot?
👉 The median is the middle value of the dataset, shown as a line inside the box.

Q7. What are whiskers in a boxplot?
👉 Whiskers represent the range of data excluding outliers.

🔹 Outliers

Q8. What are outliers?
👉 Outliers are extreme values that lie far away from other data points.

Q9. How do you detect outliers?
👉 Using IQR method:

IQR = Q3 − Q1
Lower Bound = Q1 − 1.5 × IQR
Upper Bound = Q3 + 1.5 × IQR
