# 1. Import Required Libraries
import pandas as pd
import numpy as np

# 2. Load Dataset from Web Source
url = "https://raw.githubusercontent.com/datasciencedojo/datasets/master/titanic.csv"
df = pd.read_csv(url)

print("\n--- First 5 Rows ---")
print(df.head())

# 3. Dataset Information
print("\n--- Dataset Shape ---")
print(df.shape)

print("\n--- Column Names ---")
print(df.columns)

print("\n--- Dataset Info ---")
print(df.info())

# 4. Statistical Description
print("\n--- Statistical Summary ---")
print(df.describe())

# 5. Handling Missing Values
print("\n--- Missing Values Before ---")
print(df.isnull().sum())

# Fill missing Age with mean
df['Age'].fillna(df['Age'].mean(), inplace=True)

# Fill missing Embarked with mode
df['Embarked'].fillna(df['Embarked'].mode()[0], inplace=True)

# Drop Cabin column (too many missing values)
df.drop(columns=['Cabin'], inplace=True)

print("\n--- Missing Values After ---")
print(df.isnull().sum())

# 6. Data Type Conversion
df['Age'] = df['Age'].astype(int)
df['Fare'] = df['Fare'].astype(float)

print("\n--- Data Types After Conversion ---")
print(df.dtypes)

# 7. Convert Categorical Variables to Numerical

# Convert Sex (Male=0, Female=1)
df['Sex'] = df['Sex'].map({'male': 0, 'female': 1})

# One-Hot Encoding for Embarked
df = pd.get_dummies(df, columns=['Embarked'])

print("\n--- Final Processed Data ---")
print(df.head())

# Final Check
print("\n--- Final Shape ---")
print(df.shape)

print("\n--- Final Missing Values ---")
print(df.isnull().sum())

# Optional: Save cleaned dataset
df.to_csv("cleaned_titanic.csv", index=False)
print("\nCleaned dataset saved as 'cleaned_titanic.csv'")

3. What is a dataset?
A dataset is a collection of structured data, usually in tabular form (rows and columns).
